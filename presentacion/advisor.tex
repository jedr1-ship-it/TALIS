% -------------------------------------------------------------- 8. Cierre
\begin{frame}
\frametitle{Lo que sugieren los datos y lo que queda abierto}
{\small
{\color{acc}\bfseries Sugieren}
\begin{itemize}\setlength{\itemsep}{0.6mm}
\item Frente a las expuestas con 2 años, más síntomas en las expuestas de
      bebés, +0,30 DE, al año, +0,17, y a los 3--4 años, +0,18
\item El hogar medido cambió igual a toda edad y no explica esas diferencias
\item El estrés de la madre fue igual a toda edad, pero solo predice síntomas
      en las hijas que aún eran amamantadas, el 62\% de las expuestas de bebés
      y el 4\% de las de 3--4 años
\end{itemize}
{\color{naranja}\bfseries Queda abierto}
\begin{itemize}\setlength{\itemsep}{0.6mm}
\item El estrés de la madre se mide con una pregunta de sí o no, dos años
      después
\item Qué explica el efecto en las expuestas a los 3--4 años. No pasa por el
      hogar ni por la madre, y un recuerdo de esa edad rara vez llega a la
      adolescencia; Bauer y Larkina 2014
\end{itemize}}

\nota{En varones, sin efecto en síntomas; indicios débiles en conducta y
consumo.}
\end{frame}

% -------------------------------------------------------------- 8. Cierre
\begin{frame}
\frametitle{Lo que sugieren los datos y lo que queda abierto}
{\small
{\color{acc}\bfseries Sugieren}
\begin{itemize}\setlength{\itemsep}{0.6mm}
\item Frente a las expuestas con 2 años, más síntomas en las expuestas de
      bebés, +0,30 DE, al año, +0,17, y a los 3--4 años, +0,18
\item El hogar medido cambió igual a toda edad y no explica esas diferencias
\item El estrés materno, igual a toda edad, solo se asocia a síntomas con
      lactancia activa, que baja del 62\% en bebés al 4\% a los 3--4 años
\end{itemize}
{\color{naranja}\bfseries Queda abierto}
\begin{itemize}\setlength{\itemsep}{0.6mm}
\item La medida de estrés es retrospectiva y binaria, sujeta a error de
      medición y atenuación, y la lactancia no separa contacto de leche
\item Las expuestas a los 3--4 años: no pasa por el hogar ni por la madre, y
      un recuerdo de esa edad rara vez llega a la adolescencia; Bauer y
      Larkina 2014
\end{itemize}}

\nota{En varones, sin efecto en síntomas; indicios débiles en conducta y
consumo.}
\end{frame}

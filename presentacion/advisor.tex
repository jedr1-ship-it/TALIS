% -------------------------------------------------------------- 8. Cierre
\begin{frame}
\frametitle{What the data suggest and what remains open}
{\small
{\color{acc}\bfseries What the data suggest}
\begin{itemize}\setlength{\itemsep}{0.8mm}
\item Girls exposed at 6--11 months have worse mental health than those
      exposed at age 2, by 0.30 SD on the PHQ-4.
\item The socioeconomic and family environment changed similarly at all ages
      and does not explain this difference.
\item Maternal stress is associated with worse mental health only among
      daughters who were breastfed, 62\% of those exposed at 6--11 months.
\end{itemize}
\vspace{1mm}
{\color{ocre}\bfseries What remains open}
\begin{itemize}\setlength{\itemsep}{0.8mm}
\item What explains the effect at ages 3--4, if it operates neither through
      the socioeconomic and family environment nor through maternal stress?\\[1mm]
      \textbf{Hypothesis:} at that age children can already remember
      consciously what they experience, and the memory of the earthquake
      could affect their mental health.
\end{itemize}}
\end{frame}

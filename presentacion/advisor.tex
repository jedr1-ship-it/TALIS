% -------------------------------------------------------------- 9. Cierre
\begin{frame}
\frametitle{Lo que sugieren los datos y lo que queda abierto}
{\small
{\color{acc}\bfseries Sugieren}
\begin{itemize}\setlength{\itemsep}{0.6mm}
\item Heterogeneidad marcada por la edad de exposición. El efecto aparece en
      las expuestas como lactantes y desde los 42 meses
\item El entorno familiar medido cambió igual a toda edad y no explica esas
      diferencias
\item El estrés materno se asocia a síntomas solo en díadas con lactancia
      activa, en línea con una vulnerabilidad neuroendocrina temprana
\end{itemize}
{\color{naranja}\bfseries Queda abierto}
\begin{itemize}\setlength{\itemsep}{0.6mm}
\item La medida de estrés es retrospectiva y binaria, sujeta a error de
      medición y atenuación; explica una parte pequeña de la brecha
\item La lactancia activa no separa el cortisol lácteo del contacto diádico
\item Por qué no hay efecto entre 18 y 41 meses, y si el efecto desde los 42
      meses refleja codificación de memoria episódica, que no medimos
\end{itemize}}

\nota{Solo en mujeres; en varones, nada en síntomas, conducta ni consumo.}
\end{frame}

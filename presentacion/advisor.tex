% ---------------------------------------------------------- 4. Salud mental
\begin{frame}
\frametitle{Efecto sobre la salud mental}
\centering
\includegraphics[width=\textwidth]{figuras/salud_mental.pdf}

\nota{La salud mental se mide con el PHQ-4, consistente en cuatro preguntas que
responde el adolescente sobre síntomas de ansiedad y depresión (nervios,
preocupación, pérdida de interés y decaimiento); Kroenke et al.\ 2009. Cada
punto es la diferencia entre comunas afectadas y no afectadas, con IC~al~95\%.}
\end{frame}

% --------------------------------------------------------------- 5. Hogar
\begin{frame}
\frametitle{El entorno socioeconómico y familiar no explica la brecha}
\centering
\includegraphics[width=\textwidth]{figuras/canal_hogar.pdf}

\nota{Índices de 2012 y 2017, más alto es peor, que reporta la madre o el
cuidador (el de barrio, el niño en 2017). Conflicto en casa (gritos, amenazas y
golpes al niño; peleas en el hogar y castigo físico o verbal); falta de
recursos (ingreso por persona y vivienda dañada); padre fuera del hogar; madre
sin empleo; hacinamiento (personas por dormitorio); poca estimulación
(lectura, canciones y cuentos con el niño y libros infantiles); barrio inseguro
(según el niño). Mujeres, frente a quienes tenían 2 años; IC al 95~\%, a la
derecha por bootstrap de comunas.}
\end{frame}

% ------------------------------------------------------------- 7. Varones
\begin{frame}
\frametitle{En varones, solo indicios débiles en conducta y consumo}
\centering
\includegraphics[width=\textwidth]{figuras/chicos.pdf}

\nota{Es habitual que los padres no detecten la ansiedad y la depresión de sus
hijos adolescentes; en la muestra, PHQ-4 y CBCL correlacionan 0,30; Achenbach
et al.\ 1987; De Los Reyes et al.\ 2015. En varones, ambas asociaciones son
significativas solo al 10\%.}
\end{frame}

% -------------------------------------------------------------- 8. Cierre
\begin{frame}
\frametitle{Lo que sugieren los datos y lo que queda abierto}
{\small
{\color{acc}\bfseries Lo que sugieren los datos}
\begin{itemize}\setlength{\itemsep}{0.8mm}
\item Las expuestas entre los 6 y los 11 meses tienen peor salud mental que
      las expuestas a los dos años, 0,30 DE en el PHQ-4.
\item El entorno socioeconómico y familiar cambió de forma similar en todas
      las edades y no explica esa diferencia.
\item El estrés materno solo se asocia a peor salud mental en las hijas con
      lactancia materna, el 62\% de las expuestas entre los 6 y los 11 meses.
\end{itemize}
\vspace{1mm}
{\color{naranja}\bfseries Lo que queda abierto}
\begin{itemize}\setlength{\itemsep}{0.8mm}
\item ¿Qué explica el efecto a los 3--4 años, si no opera a través del entorno
      socioeconómico y familiar ni del estrés materno?\\[1mm]
      \textbf{Hipótesis:} a esa edad el niño ya puede recordar de forma
      consciente lo que vive, y el recuerdo del terremoto podría afectar a su
      salud mental.
\end{itemize}}
\end{frame}

\begin{frame}
\frametitle{Cómo lo medimos}
\vspace{3mm}
\begin{equation*}
Y_{imc} = \sum_{a \neq 2} \beta_a \,(\mathrm{Edad}_{ai} \times \mathrm{Afectada}_m)
 + \sum_{a \neq 2} \gamma_a\, \mathrm{Edad}_{ai} + \psi_m + \theta_c + X_i'\delta + \varepsilon_{imc}
\end{equation*}

\begin{center}
$H_0:\ \beta_a = 0$ para toda edad $a$

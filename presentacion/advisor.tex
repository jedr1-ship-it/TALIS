% -------------------------------------------------------------- 9. Cierre
\begin{frame}
\frametitle{Lo que sugieren los datos y lo que queda abierto}
\vspace{2mm}
{\color{acc}\bfseries Sugieren}
\begin{itemize}\setlength{\itemsep}{1mm}
\item El efecto depende de la edad de exposición. Aparece en las expuestas
      de bebés y desde los 42 meses
\item El hogar que medimos cambió igual a toda edad y no explica esas
      diferencias
\item El estrés de la madre se asocia a síntomas solo en las hijas que
      tomaban pecho, en línea con una vulnerabilidad neuroendocrina temprana
\end{itemize}
\vspace{2mm}
{\color{naranja}\bfseries Queda abierto}
\begin{itemize}\setlength{\itemsep}{1mm}
\item La medida de estrés es gruesa y explica una parte pequeña de la brecha
\item Por qué no hay efecto entre 18 y 41 meses. Las ventanas de
      sensibilidad parecen más complejas de lo que capturan los modelos
\item Si lo que aparece desde los 42 meses es memoria explícita, que no
      medimos
\end{itemize}

\nota{Solo en chicas; en chicos, nada en síntomas, conducta ni consumo.}
\end{frame}

% ------------------------------------------------------------ 8. Hallazgos
\begin{frame}
\frametitle{Findings}
\vspace{4mm}
\begin{itemize}\setlength{\itemsep}{4mm}
\item Girls exposed at 6--11 months have worse mental health than those
      exposed at older ages, by 0.18 SD on the PHQ-4; boys show no difference.
\item The socioeconomic and family environment did not worsen more for those
      aged 6--11 months and does not explain this difference.
\item The effect at 6--11 months is concentrated among girls who were
      breastfed on 27-F, 62\% of that age group.
\item Maternal stress reported in 2012 is associated with worse mental health
      only among breastfed daughters, but accounts for about 0.01 SD of the effect.
\end{itemize}
\end{frame}

% ------------------------------------------------- 9. Lo que queda abierto
\begin{frame}
\frametitle{Open questions}
\vspace{4mm}
\begin{itemize}\setlength{\itemsep}{4mm}
\item Why do symptoms appear only among girls?\\[3mm]
      \textbf{Hypothesis:} early stress leaves a latent vulnerability that
      surfaces in adolescence, when the gender gap in depression opens
      \cita{(McCrory and Viding, 2015; Salk, Hyde, and Abramson, 2017)}.
\item What does breastfeeding transmit, if the stress the mother reports
      accounts for little of the effect?\\[3mm]
      \textbf{Hypothesis:} the mother's acute stress response in the months
      after the earthquake, which a report two years later captures only in
      part.
\end{itemize}
\end{frame}

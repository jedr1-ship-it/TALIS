\begin{frame}
\frametitle{¿Importa la edad a la que llega el shock?}
\vspace{3mm}
\begin{equation*}
Y_{imc} = \sum_{a \neq 2} \beta_a \,(\mathrm{Edad}_{ai} \times \mathrm{Afectada}_m)
 + \sum_{a \neq 2} \gamma_a\, \mathrm{Edad}_{ai} + \psi_m + \theta_c + X_i'\delta + \varepsilon_{imc}
\end{equation*}
\vspace{1mm}
\begin{center}
$H_0:\ \beta_a = 0$ para toda edad $a$
\end{center}
\vspace{1mm}
{\small
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{@{}l l@{}}
{\color{muted}$Y_{imc}$} & Salud mental a los 15--18 años, PHQ-4 \\
{\color{muted}$\mathrm{Edad}_{ai}$} & Edad al terremoto: 6--11 meses, 1 año o 3--4 años; referencia, 2 años \\
{\color{muted}$\mathrm{Afectada}_m$} & Comuna en la zona de daño destructivo \\
{\color{muted}$\beta_a$} & Diferencia con quienes tenían 2 años, comunas afectadas menos no afectadas \\
{\color{muted}$\psi_m$, $\theta_c$} & Efectos fijos de comuna y de cohorte de nacimiento \\
{\color{muted}$X_i$} & Controles del niño y del hogar \\
\end{tabular}}
